# List

- List.BasicItem are used to display simple rows of information: prefix, title, subtitle, suffix
- List.Item are used to display rows of information: Basic Item, leading status and customize views

## Examples

### Basic Item

```jsx live
function BasicItem() {
  const Section = ({ children }) => {
    const theme = useTheme();
    return (
      <SectionHeading
        fontWeight="semi-bold"
        fontSize="xlarge"
        intent="primary"
        style={{
          marginVertical: theme.space.medium,
        }}
        text={children}
      />
    );
  };

  const BasicItemData = [
    { title: 'List Item' },
    { title: 'List Item with suffix', suffix: <Switch checked /> },
    {
      title: 'List item with prefix and suffix',
      suffix: 'checkmark',
      prefix: <Avatar title="ML" />,
    },
    {
      title: 'List item with value',
      suffix: <Typography.Body variant="small">$500.00</Typography.Body>,
    },
    { disabled: true, title: 'Disabled list item', suffix: 'checkmark' },
  ];

  return (
    <>
      <Section>Item inside a View</Section>
      <List.BasicItem title="Basic Item" suffix="checkmark" />
      <List.Item title="Item" suffix="checkmark" />
      <FlatList
        ItemSeparatorComponent={Divider}
        data={BasicItemData}
        renderItem={({ item }) => {
          return (
            <List.BasicItem
              {...item}
              style={{ marginVertical: theme.space.xsmall }}
            />
          );
        }}
      />
    </>
  );
}
```
